// Formatter Helpers
const fmtCr = (n) => {
  const sign = n < 0 ? '-' : '';
  const abs = Math.abs(n);
  if (abs >= 1e7) return sign + '₹' + (abs / 1e7).toFixed(2) + ' Cr';
  if (abs >= 1e5) return sign + '₹' + (abs / 1e5).toFixed(2) + ' L';
  if (abs >= 1e3) return sign + '₹' + (abs / 1e3).toFixed(1) + 'K';
  return sign + '₹' + Math.round(abs);
};

const fmtINR = (n) => {
  return '₹' + Math.round(n).toLocaleString('en-IN');
};

const fmtInt = (n) => {
  return Math.round(n).toLocaleString('en-IN');
};

const parseDH = (str) => {
  const chips = [];
  const s = (str || '').toString();
  for (const ch of s) {
    if (/[A-Za-z]/.test(ch) && DH_COLORS[ch.toUpperCase()]) {
      chips.push({ ch: ch.toUpperCase(), color: DH_COLORS[ch.toUpperCase()] });
    }
  }
  return chips.slice(0, 32);
};
